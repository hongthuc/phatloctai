    let currentBanner = 0;
    let bannerTimer = null;

    function updateBannerDisplay() {
      const slides = document.querySelectorAll('#hero-slider .banner-slide');
      slides.forEach((slide, idx) => {
        if (idx === currentBanner) {
          slide.classList.add('is-active');
          slide.classList.remove('is-inactive');
        } else {
          slide.classList.remove('is-active');
          slide.classList.add('is-inactive');
        }
      });

      [0, 1].forEach(idx => {
        const dot = document.getElementById('dot-' + idx);
        if (!dot) return;
        if (idx === currentBanner) {
          dot.classList.remove('w-2.5', 'bg-white/50');
          dot.classList.add('w-6', 'bg-brand-gold');
        } else {
          dot.classList.remove('w-6', 'bg-brand-gold');
          dot.classList.add('w-2.5', 'bg-white/50');
        }
      });
    }

    function nextBanner() {
      currentBanner = (currentBanner + 1) % 2;
      updateBannerDisplay();
      resetBannerAutoSlide();
    }

    function prevBanner() {
      currentBanner = (currentBanner + 1) % 2;
      updateBannerDisplay();
      resetBannerAutoSlide();
    }

    function goToBanner(index) {
      currentBanner = index;
      updateBannerDisplay();
      resetBannerAutoSlide();
    }

    function resetBannerAutoSlide() {
      if (bannerTimer) clearInterval(bannerTimer);
      bannerTimer = setInterval(() => {
        currentBanner = (currentBanner + 1) % 2;
        updateBannerDisplay();
      }, 5000);
    }

    // Hỗ trợ cử chỉ vuốt trái / phải trên điện thoại cho Banner
    function initBannerSwipe() {
      const slider = document.getElementById('hero-slider');
      if (!slider) return;
      let startX = 0;
      slider.addEventListener('touchstart', (e) => {
        startX = e.touches[0].clientX;
      }, { passive: true });
      slider.addEventListener('touchend', (e) => {
        const endX = e.changedTouches[0].clientX;
        if (Math.abs(startX - endX) > 40) {
          nextBanner();
        }
      }, { passive: true });
    }

      const box = document.getElementById('toast-box');
      const msg = document.getElementById('toast-message');

    function submitQuoteForm(event) {

      try {
        const name = document.getElementById('customer-name').value.trim();
        const phone = document.getElementById('customer-phone').value.trim();
        const service = document.getElementById('customer-service').value;

        if (!name || !phone) {
          showToast('Vui lòng nhập đầy đủ Họ tên và Số điện thoại để nhận báo giá.');

        showToast('Đã tiếp nhận thông tin. Vui lòng liên hệ trực tiếp Hotline 0919.353.309 nếu cần gấp.');
